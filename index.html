function parseSuperbidTexto(texto) { const linhas = texto .replace(/Ícone[^\n\r]*/gi, "") .split(/\r?\n/) .map(l => l.trim()) .filter(l => l && l !== "-"); const encontrados = []; const regexCabecalhoCard = /^(\d+\s+pra[çc]as\s*\||Pra[çc]a\s+[ÚU]nica\s*\||\d{2}\/\d{2}\s+-\s+\d{2}:\d{2}$)/i; const regexCidadeUF = /([A-ZÀ-Ú][a-zà-ú]+(?:\s+(?:de|da|do|dos|das|[A-ZÀ-Ú][a-zà-ú]+))*\s*-\s*[A-Z]{2})(?:\s*-\s*[A-Z]{2})?$/; const regexOutraMoto = /^(HONDA|YAMAHA|SUZUKI|KAWASAKI|BMW|SHINERAY|DAFRA|HAOJUE|TRIUMPH|ROYAL|BAJAJ|SUNDOWN|CG\s+\d+|SUCATA|MOTOCICLETA)\b/i; for (let i = 0; i < linhas.length; i++) { const mLote = linhas[i].match(/^Lote\s+(\d+)$/i); if (!mLote) continue; const numLote = mLote[1]; const linhaData = i > 0 ? linhas[i - 1] : ""; const linhaTituloRaw = i + 1 < linhas.length ? linhas[i + 1] : ""; if (!linhaTituloRaw) continue; if (/^(SUCATA|PE[ÇC]AS|[ÓO]RG[ÃA]O|KIT\b|VESTU[ÁA]RIO)/i.test(linhaTituloRaw)) continue; let tituloComAno = linhaTituloRaw; let local = ""; const mCidadeColada = linhaTituloRaw.match(regexCidadeUF); if (mCidadeColada) { local = mCidadeColada[1].replace(/\s*-\s*/, " - ").trim(); tituloComAno = linhaTituloRaw.slice(0, mCidadeColada.index).trim(); } else { const mCidadeSemUF = linhaTituloRaw.match(/(São José do Rio Preto|Campo Mourão)$/i); if (mCidadeSemUF) { local = mCidadeSemUF[1]; tituloComAno = linhaTituloRaw.slice(0, mCidadeSemUF.index).trim(); } } const bloco = []; for (let j = i + 2; j < Math.min(linhas.length, i + 8); j++) { const atual = linhas[j]; const proxima = j + 1 < linhas.length ? linhas[j + 1] : ""; const depois = j + 2 < linhas.length ? linhas[j + 2] : ""; // Para antes do título/data do próximo card if (/^Lote\s+\d+$/i.test(atual) || /^Lote\s+\d+$/i.test(proxima) || /^Lote\s+\d+$/i.test(depois)) break; if (regexCabecalhoCard.test(atual) || /^Perguntas e respostas/i.test(atual)) break; // Ignora repetições de nome de moto (alt de imagens ou vazamento do próximo card) if (regexOutraMoto.test(atual)) continue; if (!local && regexCidadeUF.test(atual)) { local = atual.match(regexCidadeUF)[1].replace(/\s*-\s*/, " - ").trim(); continue; } if (!local && /^(São José do Rio Preto|Campo Mourão)$/i.test(atual)) { local = atual; continue; } if (!bloco.includes(atual)) bloco.push(atual); } const blocoStr = [linhaData, tituloComAno, ...bloco].join(" | "); let ano = 0; const mAno4 = blocoStr.match(/\b(199\d|20[0-2]\d)(?:\/(199\d|20[0-2]\d))?\b/); if (mAno4) { ano = parseInt(mAno4[2] || mAno4[1], 10); } else { const mAno2 = blocoStr.match(/\b(\d{2})\/(\d{2})\b/); if (mAno2 && !blocoStr.includes(mAno2[0] + " -")) { const yy = parseInt(mAno2[2], 10); ano = yy <= 30 ? 2000 + yy : 1900 + yy; } } const valores = Array.from(blocoStr.matchAll(/R\$\s*([\d\.]+)(?:,(\d{2}))?/g)) .map(m => parseFloat(m[1].replace(/\./g, "") + "." + (m[2] || "00"))); const lance = valores.length ? Math.min(...valores) : 0; let dataLeilao = ""; const mData = linhaData.match(/(\d{2})\/(\d{2})/); if (mData) { const mes = parseInt(mData[2], 10); const anoData = mes < 6 ? 2027 : 2026; dataLeilao = `${anoData}-${mData[2]}-${mData[1]}`; } const tituloLimpo = tituloComAno .replace(/^MOTOCICLETA\s+/i, "") .replace(/^Uma Motocicleta\s+/i, "") .replace(/,?\s*\b(199\d|20[0-2]\d)(?:\/(199\d|20[0-2]\d))?\b.*$/i, "") .replace(/,?\s*\b\d{2}\/\d{2}\b.*$/i, "") .replace(/\s+/g, " ") .trim(); const origem = detectarOrigem({ titulo: tituloLimpo, descricao: blocoStr, comitente: "" }); encontrados.push({ leiloeiro: "Superbid", lote: numLote, titulo: tituloLimpo || tituloComAno, ano: ano, km: 0, lance: lance, fipe: 0, comitente: "Superbid Exchange (ver edital do lote)", descricao: [linhaData, ...bloco].filter(Boolean).join(" • "), origem: origem, local: local || "Brasil", data_leilao: dataLeilao, url: `https://www.superbid.net/busca?searchTerm=${encodeURIComponent(tituloLimpo || numLote)}` }); } return encontrados; }
